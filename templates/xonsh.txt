def __zoxide_bin() -> str:
    """Finds and returns the location of the zoxide binary."""
    zoxide = typing.cast(str, xonsh.environ.locate_binary("zoxide"))
    if zoxide is None:
        zoxide = "zoxide"
    return zoxide


def __zoxide_env() -> dict[str, str]:
    """Returns the current environment."""
    return builtins.__xonsh__.env.detype()  # type: ignore  # pylint:disable=no-member


def __zoxide_pwd() -> str:
    """pwd based on the value of _ZO_RESOLVE_SYMLINKS."""

{%- endif %}
    return pwd


def __zoxide_cd(path: str | bytes | None = None) -> None:
    """cd + custom logic based on the value of _ZO_ECHO."""
    if path is None:
        args = []
    elif isinstance(path, bytes):
        args = [path.decode("utf-8")]
    else:
        args = [path]
    _, exc, _ = xonsh.dirstack.cd(args)
    if exc is not None:
        raise RuntimeError(exc)

class ZoxideSilentException(Exception):
    """Exit without complaining."""


def __zoxide_errhandler(
    func: typing.Callable[[list[str]], None],
) -> typing.Callable[[list[str]], int]:
    """Print exception and exit with error code 1."""

    def wrapper(args: list[str]) -> int:
        try:
            func(args)
            return 0
        except ZoxideSilentException:
            return 1
        except Exception as exc:  # pylint: disable=broad-except
            print(f"zoxide: {exc}", file=sys.stderr)
            return 1

    return wrapper

@__zoxide_errhandler
def __zoxide_z(args: list[str]) -> None:
    """Jump to a directory using only keywords."""
    if args == []:
        __zoxide_cd()
    elif args == ["-"]:
        __zoxide_cd("-")
    elif len(args) == 1 and os.path.isdir(args[0]):
        __zoxide_cd(args[0])
    else:
        try:
            zoxide = __zoxide_bin()
            cmd = subprocess.run(
                [zoxide, "query", "--exclude", __zoxide_pwd(), "--"] + args,
                check=True,
                env=__zoxide_env(),
                stdout=subprocess.PIPE,
            )
        except subprocess.CalledProcessError as exc:
            raise ZoxideSilentException() from exc

        result = cmd.stdout[:-1]
        __zoxide_cd(result)


@__zoxide_errhandler
def __zoxide_zi(args: list[str]) -> None:
    """Jump to a directory using interactive search."""
    try:
        zoxide = __zoxide_bin()
        cmd = subprocess.run(
            [zoxide, "query", "-i", "--"] + args,
            check=True,
            env=__zoxide_env(),
            stdout=subprocess.PIPE,
        )
    except subprocess.CalledProcessError as exc:
        raise ZoxideSilentException() from exc

    result = cmd.stdout[:-1]
    __zoxide_cd(result)

# Interactive completions for prompt_toolkit.
#

# Keep manually assigned aliases recognizable after reinitialization.
setattr(__zoxide_z, "__zoxide_complete", True)


def __zoxide_complete_command(
    text: str, parser: typing.Any
) -> tuple[str, list[str]] | None:
    """Recognize a standalone command without evaluating shell expressions."""
    if not text.endswith(" ") or "\n" in text or "\r" in text:
        return None
    context = parser.parse(text, len(text))
    if context is None or context.command is None:
        return None
    current = context.command
    name = current.args[0].raw_value if current.args else ""
    aliases = builtins.aliases  # type: ignore # pylint:disable=no-member
    if (
        len(current.args) < 2
        or current.arg_index != len(current.args)
        or name not in aliases
        or not getattr(
            getattr(aliases[name], "func", aliases[name]), "__zoxide_complete", False
        )
    ):
        return None
    pattern = (
        r"\s*" + r"\s+".join(re.escape(arg.raw_value) for arg in current.args) + r"\s+"
    )
    if re.fullmatch(pattern, text) is None:
        return None
    args = []
    try:
        for arg in current.args[1:]:
            if arg.opening_quote:
                value = ast.literal_eval(arg.raw_value)
                if not isinstance(value, str):
                    raise ValueError("not a string")
                if "$" in value or value.startswith("~"):
                    raise ValueError("not a literal keyword")
            else:
                if any(char in arg.value for char in "\"'$@();|<>&`\\*?[]{}~"):
                    raise ValueError("not a literal keyword")
                value = arg.value
            args.append(value)
    except (SyntaxError, ValueError):
        return None
    prefix = text[: len(text) - len(text.lstrip())] + name
    return prefix, args


if "__zoxide_keybindings" not in globals():

    @builtins.events.on_ptk_create  # type: ignore  # pylint:disable=no-member
    def __zoxide_keybindings(bindings: typing.Any, **_kwargs: typing.Any) -> None:
        """Keep native Tab completion, and use fzf after a keyword and a space."""
        # These dependencies are only needed by the prompt_toolkit shell.
        # pylint: disable=import-outside-toplevel
        from prompt_toolkit.application import (  # type: ignore[import-not-found, unused-ignore] # pylint:disable=import-error
            get_app,
            run_in_terminal,
        )
        from prompt_toolkit.document import Document  # type: ignore[import-not-found, unused-ignore] # pylint:disable=import-error
        from prompt_toolkit.filters import (  # type: ignore[import-not-found, unused-ignore] # pylint:disable=import-error
            Condition,
            emacs_insert_mode,
            has_completions,
            vi_insert_mode,
        )
        from xonsh.parsers.completion_context import (  # type: ignore # pylint:disable=import-error
            CompletionContextParser,
        )

        parser = CompletionContextParser()
        pending = False

        def enabled() -> bool:
            document = get_app().current_buffer.document
            return (
                not pending
                and document.is_cursor_at_the_end
                and not has_completions()
                and __zoxide_complete_command(document.text, parser) is not None
            )

        def complete(event: typing.Any) -> None:
            """Suspend the prompt while fzf owns the terminal."""
            nonlocal pending
            buffer = event.current_buffer
            document = buffer.document
            parsed = __zoxide_complete_command(document.text, parser)
            if parsed is None:
                return
            prefix, args = parsed
            pending = True

            def select() -> None:
                nonlocal pending
                try:
                    result = subprocess.run(
                        [
                            __zoxide_bin(),
                            "query",
                            "-i",
                            "--exclude",
                            __zoxide_pwd(),
                            "--",
                        ]
                        + args,
                        check=False,
                        env=__zoxide_env(),
                        stdout=subprocess.PIPE,
                    )
                    if result.returncode != 0 or not result.stdout.endswith(b"\n"):
                        return
                    path = result.stdout[:-1].decode("utf-8")
                    if path and buffer.document == document:
                        # Quoted subprocess strings still expand environment variables.
                        text = f"{prefix} @({path!r}) "
                        buffer.document = Document(text, len(text))
                except (OSError, UnicodeError, RuntimeError) as exc:
                    print(f"zoxide: {exc}", file=sys.stderr)
                finally:
                    pending = False

            run_in_terminal(select)

        bindings.add(
            "c-i", filter=Condition(enabled) & (emacs_insert_mode | vi_insert_mode)
        )(complete)
